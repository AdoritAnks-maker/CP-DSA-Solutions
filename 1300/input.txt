#include<bits/stdc++.h>
using namespace std;

int main(){
    ios::sybc_with_stdio(false);
    cin.tie(NULL);
    
    int n ;
    cin >> n;
    vector<int>arr(n+1);
    for(int i  = 1 ;i<=n;i++){
        cin >> arr[i];
    }
    int count  = 0;
    sort(arr.begin() , arr.end());
    for(int i  = 1 ;i<=n ;i++){
        if(i!=arr[i]){
            count++;
        }
    }
    cout << count << endl;
}
